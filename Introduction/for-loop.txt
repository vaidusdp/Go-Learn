package main

import (
	"fmt"
)

func main() {

	for i := 1; i <= 10; i++ {
		if i%2 == 0 {
			continue
		}
		fmt.Println(i)
		if i == 5 {
			break
		}
	}

	numbers := []int{1, 2, 3, 4, 5, 6}
	for index, value := range numbers {
		fmt.Printf("Index: %d, Value: %v\n", index, value)
		// %d and %v are similar
	}

	rows := 5

	for i:=1; i<=rows; i++ {
		for j:=1; j<=i; j++ {
			fmt.Print("*")
		}
		fmt.Println()
	}

	for i := range 10 {
		i++
		fmt.Println(10-i);
	}
}
