package main

import "fmt"

func main() {
	var arr [5]int
	fmt.Println(arr)

	arr[4] = 20
	fmt.Println(arr)

	fruits := [4] string{"A", "B", "G", "P"}
	fmt.Println(fruits);

	copiedFruits := fruits
	copiedFruits[0] = "X"

	fmt.Println(fruits);
	fmt.Println(copiedFruits);

	for i := 0; i<len(arr); i++ {
		fmt.Println(i, ":", arr[i])
	}

	var matrix [3][3]int = [3][3]int{
		{1, 2, 3},
		{4, 5, 6},
		{7, 8, 9},
	}

	fmt.Println(matrix);

	orgArr := [3]int {1, 2, 3}
	var add *[3]int 

	add = &orgArr

	fmt.Println(orgArr)
	fmt.Println(add)


}
